import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.chrome.ChromeOptions;
import org.openqa.selenium.support.ui.WebDriverWait;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.Select;

import java.time.Duration;

public class FirstTest {

    public static void main(String[] args) throws InterruptedException {

        ChromeOptions options = new ChromeOptions();
        options.addArguments("--remote-allow-origins=*");

        WebDriver driver = new ChromeDriver(options);
        WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));

        try {

            // starting my driver and chrome will load
            driver.get("http://localhost:5173");
            driver.manage().window().maximize();
            Thread.sleep(2000);

            // Go to Register page to register the user
            wait.until(ExpectedConditions.elementToBeClickable(By.linkText("Register"))).click();
            Thread.sleep(1500);

            // Register User with the rifht details
            wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("name"))).sendKeys("Agrim Verma");
            Thread.sleep(800);

            driver.findElement(By.id("email")).sendKeys("agrim@thapar.edu");
            Thread.sleep(800);

            driver.findElement(By.id("Password")).sendKeys("abc123");
            Thread.sleep(800);

            driver.findElement(By.id("confirmPassword")).sendKeys("abc123");
            Thread.sleep(1000);

            driver.findElement(By.id("registerBtn")).click();
            Thread.sleep(2000);

            //  Waiting for Login Page to show again
            wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("loginEmail")));
            Thread.sleep(1000);

            // Makinf a wrong Login attempt
            driver.findElement(By.id("loginEmail")).sendKeys("wrong@thapar.edu");
            Thread.sleep(800);

            driver.findElement(By.id("loginPassword")).sendKeys("abc123");
            Thread.sleep(1000);

            driver.findElement(By.id("loginBtn")).click();
            Thread.sleep(2500);

            System.out.println("Wrong login attempt - displayed error");

            // Refresh page to reset React state cleanly cuz earlier i had the wrong details in
            driver.navigate().refresh();

            wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("loginEmail")));
            Thread.sleep(1500);

            // Now putting in the Correct Login details
            driver.findElement(By.id("loginEmail")).sendKeys("agrim@thapar.edu");
            Thread.sleep(800);

            driver.findElement(By.id("loginPassword")).sendKeys("abc123");
            Thread.sleep(1000);

            driver.findElement(By.id("loginBtn")).click();
            Thread.sleep(2000);

            wait.until(ExpectedConditions.urlContains("/home"));
            System.out.println("Successfully logged in ");
            Thread.sleep(2000);

            // Demonstrate Home page functionality of yk segregating based on the tags
            Select categoryDropdown = new Select(
                    wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("categoryFilter")))
            );

            categoryDropdown.selectByVisibleText("Internship");
            Thread.sleep(2000);

            System.out.println("Filtered Internship opportunities");

            // Logout of the page and phirse login pe aajao
            wait.until(ExpectedConditions.elementToBeClickable(By.className("nav-logout"))).click();
            Thread.sleep(2000);

            System.out.println("Logout successful - back to where we started hehehe");
            Thread.sleep(2000);

        } catch (Exception e) {
            System.out.println("Automation Failed - error came");
            e.printStackTrace();
        } finally {
            driver.quit();
        }
    }
}
